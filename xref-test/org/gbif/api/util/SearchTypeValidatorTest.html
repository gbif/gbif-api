243   @ParameterizedTest
244   @MethodSource("data")
245   public void testValidateFine(SearchParameter parameter, String arg, boolean valid, boolean numericRange, boolean dateRange) {
246     try {
247       SearchTypeValidator.validate(parameter, arg);
248       if (!valid) {
249         fail(arg + " supposed to be an invalid value for parameter " + parameter);
250       }
251     } catch (IllegalArgumentException e) {
252       if (valid) {
253         fail(arg + " supposed to be a valid value for parameter " + parameter + " " + e.getMessage());
254       }
255     }
256 
257     // test isNumericRange
258     assertEquals(numericRange, SearchTypeValidator.isNumericRange(arg), "Wrong isNumericRange parsing of value " + arg);
259 
260     // test isDateRange
261     assertEquals(dateRange, SearchTypeValidator.isDateRange(arg), "Wrong isDateRange parsing of value " + arg);
262   }

294   })
295   public void testRestrictedParametersRejectInjection(String name) {
296     List<SearchParameter> parameters = parametersNamed(name);
297     assertFalse(parameters.isEmpty(), "No occurrence or event search parameter named " + name);
298 
299     for (SearchParameter parameter : parameters) {
300       assertThrows(
301         IllegalArgumentException.class,
302         () -> SearchTypeValidator.validate(parameter, "'"),
303         "Value ' should be invalid for " + parameter.getClass().getSimpleName() + "." + name);
304     }
305   }
306 
307   private static List<SearchParameter> parametersNamed(String name) {
308     return Stream.concat(
309         Stream.of(OccurrenceSearchParameter.values()),
310         Stream.of(EventSearchParameter.values()))
311       .filter(p -> name.equals(p.name()))
312       .filter(p -> String.class.equals(p.type()))
313       .collect(Collectors.toList());
314   }
